cmake_minimum_required(VERSION 3.25)
project(YideckAudio VERSION 1.1.0 LANGUAGES CXX RC)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
option(YIDECK_ENABLE_ASIO "Build the licensed single-driver ASIO adapter" OFF)
set(YIDECK_RUNTIME_DIR "${CMAKE_CURRENT_SOURCE_DIR}/bin" CACHE PATH "Product executable output directory")
set(YIDECK_TEST_RUNTIME_DIR "${CMAKE_CURRENT_SOURCE_DIR}/test-bin" CACHE PATH "Native test output directory")
set(DEPS "${CMAKE_CURRENT_SOURCE_DIR}/../third_party" CACHE PATH "Pinned audio dependency sources")
set(ASIO_INCLUDE "" CACHE PATH "Separately licensed ASIO SDK interface headers")
set(VSTSDK "${DEPS}/vst3sdk")
set(RTAUDIO "${DEPS}/rtaudio-06f97b633c06ea747d6478f0e87b6e1178104511")
set(VST2SDK "${DEPS}/vst2sdk-339d4f31590bf77c0d0d248e09a380ac6285e069")
set(SMTG_ENABLE_VST3_PLUGIN_EXAMPLES OFF CACHE BOOL "" FORCE)
set(SMTG_ENABLE_VST3_HOSTING_EXAMPLES OFF CACHE BOOL "" FORCE)
set(SMTG_ENABLE_VSTGUI_SUPPORT OFF CACHE BOOL "" FORCE)
set(SMTG_RUN_VST_VALIDATOR OFF CACHE BOOL "" FORCE)
set(SMTG_CREATE_PLUGIN_LINK OFF CACHE BOOL "" FORCE)
set(SMTG_CREATE_MODULE_INFO OFF CACHE BOOL "" FORCE)
set(SMTG_USE_STATIC_CRT ON CACHE BOOL "" FORCE)
add_subdirectory("${VSTSDK}" "${CMAKE_BINARY_DIR}/vst3sdk" EXCLUDE_FROM_ALL)
# RtAudio owns WASAPI. The optional ASIO adapter loads only the explicitly selected
# driver through the pinned SDK interface, never RtAudio's all-driver probe.
add_library(yideck_rtaudio STATIC "${RTAUDIO}/RtAudio.cpp")
target_compile_definitions(yideck_rtaudio PRIVATE __WINDOWS_WASAPI__ NOMINMAX WIN32_LEAN_AND_MEAN)
target_include_directories(yideck_rtaudio PUBLIC "${RTAUDIO}" PRIVATE "${RTAUDIO}/include")
target_link_libraries(yideck_rtaudio PUBLIC ole32 winmm ksuser mfplat mfuuid wmcodecdspuuid)
add_library(yideck_plugins STATIC src/plugins.cpp "${VSTSDK}/public.sdk/source/vst/hosting/module_win32.cpp" src/vst3_plugprovider.cpp "${VSTSDK}/public.sdk/source/common/memorystream.cpp")
target_include_directories(yideck_plugins PUBLIC src deps "${VSTSDK}" "${VST2SDK}/include")
target_compile_definitions(yideck_plugins PUBLIC NOMINMAX WIN32_LEAN_AND_MEAN UNICODE _UNICODE)
target_compile_options(yideck_plugins PUBLIC /utf-8 /EHsc /Zc:char8_t-)
target_link_libraries(yideck_plugins PUBLIC sdk_hosting crypt32 user32 ole32)
add_executable(Yideck.Audio.Engine src/engine.cpp)
target_link_libraries(Yideck.Audio.Engine PRIVATE yideck_plugins yideck_rtaudio)
if(YIDECK_ENABLE_ASIO)
  if(NOT EXISTS "${ASIO_INCLUDE}/asio.h")
    message(FATAL_ERROR "ASIO requires separately licensed headers via ASIO_INCLUDE")
  endif()
  target_compile_definitions(Yideck.Audio.Engine PRIVATE YIDECK_ENABLE_ASIO=1)
  target_include_directories(Yideck.Audio.Engine PRIVATE "${ASIO_INCLUDE}")
  target_link_libraries(Yideck.Audio.Engine PRIVATE advapi32)
  add_executable(Yideck.Test.Asio tests/test_asio.cpp)
  target_compile_definitions(Yideck.Test.Asio PRIVATE YIDECK_ASIO_TEST=1)
  target_include_directories(Yideck.Test.Asio PRIVATE src "${ASIO_INCLUDE}")
  target_link_libraries(Yideck.Test.Asio PRIVATE yideck_plugins advapi32)
  set_target_properties(Yideck.Test.Asio PROPERTIES RUNTIME_OUTPUT_DIRECTORY_RELEASE "${YIDECK_TEST_RUNTIME_DIR}")
endif()
add_executable(Yideck.Audio.Scan src/scan.cpp)
target_link_libraries(Yideck.Audio.Scan PRIVATE yideck_plugins)
foreach(t Yideck.Audio.Engine Yideck.Audio.Scan)
  set(AUDIO_FILENAME "${t}.exe")
  set(AUDIO_INTERNAL_NAME "${t}")
  if(t STREQUAL "Yideck.Audio.Engine")
    set(AUDIO_DESCRIPTION "易音音频机架引擎")
  else()
    set(AUDIO_DESCRIPTION "易音独立音频插件扫描器")
  endif()
  get_filename_component(AUDIO_ICON "${CMAKE_CURRENT_SOURCE_DIR}/../assets/yiyin.ico" ABSOLUTE)
  configure_file(resources/audio.rc.in "${CMAKE_BINARY_DIR}/${t}.rc" @ONLY)
  target_sources(${t} PRIVATE "${CMAKE_BINARY_DIR}/${t}.rc")
  set_target_properties(${t} PROPERTIES RUNTIME_OUTPUT_DIRECTORY_RELEASE "${YIDECK_RUNTIME_DIR}" RUNTIME_OUTPUT_DIRECTORY_DEBUG "${YIDECK_RUNTIME_DIR}-debug")
endforeach()
